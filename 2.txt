```go
		env("DB_HOST", "localhost"),
		env("DB_USER", "postgres"),
		env("DB_PASSWORD", "postgres"),

		env("DB_PORT", "5432"),

		DB, err = gorm.Open(postgres.Open(dsn), &gorm.Config{})
		if err == nil {
			break
		}

	if err := DB.AutoMigrate(&model.User{}, &model.Todo{}); err != nil {
		log.Fatal("迁移失败:", err)
	}
	log.Println("数据库就绪")
}

	}
	return d
```

### 3. 用数据库重写 Store

**backend/internal/store/store.go**
```go
package store

import (
	"todo-app/internal/db"
	"todo-app/internal/model"
)

type Store struct{}

func New() *Store { return &Store{} }

func (s *Store) ListTodos(userID uint) []model.Todo {
	var todos []model.Todo
	db.DB.Where("user_id = ?", userID).Order("id desc").Find(&todos)
	return todos
}

func (s *Store) CreateTodo(userID uint, title string) model.Todo {
	t := model.Todo{UserID: userID, Title: title}
	db.DB.Create(&t)
	return t
}

func (s *Store) ToggleTodo(userID, id uint) (model.Todo, bool) {
	var t model.Todo
	if err := db.DB.Where("id = ? AND user_id = ?", id, userID).First(&t).Error; err != nil {
		return model.Todo{}, false
	}
	t.Done = !t.Done
	db.DB.Save(&t)
	return t, true
}

func (s *Store) DeleteTodo(userID, id uint) bool {
	res := db.DB.Where("id = ? AND user_id = ?", id, userID).Delete(&model.Todo{})
	return res.RowsAffected > 0
}

// ---------- 用户 ----------

func (s *Store) CreateUser(username, hashedPassword string) (model.User, error) {
	u := model.User{Username: username, Password: hashedPassword}
	err := db.DB.Create(&u).Error
	return u, err
}

func (s *Store) FindUserByUsername(username string) (model.User, bool) {
	var u model.User
	if err := db.DB.Where("username = ?", username).First(&u).Error; err != nil {
		return model.User{}, false
	}
	return u, true
}
```

## 二、JWT 认证

### 1. 认证工具

**backend/internal/auth/auth.go**
```go
package auth

import (
	"errors"
	"os"
	"time"

	"github.com/golang-jwt/jwt/v5"
	"golang.org/x/crypto/bcrypt"
)

var secret = []byte(env("JWT_SECRET", "change-me-in-prod"))

func env(k, d string) string {
	if v := os.Getenv(k); v != "" {
		return v
	}
	return d
}

// 密码哈希
func HashPassword(pw string) (string, error) {
	b, err := bcrypt.GenerateFromPassword([]byte(pw), bcrypt.DefaultCost)
	return string(b), err
}

func CheckPassword(hash, pw string) bool {
	return bcrypt.CompareHashAndPassword([]byte(hash), []byte(pw)) == nil
}

// 生成 JWT
func GenerateToken(userID uint, username string) (string, error) {
	claims := jwt.MapClaims{
		"uid":      userID,
		"username": username,
		"exp":      time.Now().Add(72 * time.Hour).Unix(),
		"iat":      time.Now().Unix(),
	}
	return jwt.NewWithClaims(jwt.SigningMethodHS256, claims).SignedString(secret)
}

// 解析 JWT
func ParseToken(tokenStr string) (uint, string, error) {
	token, err := jwt.Parse(tokenStr, func(t *jwt.Token) (interface{}, error) {
		if _, ok := t.Method.(*jwt.SigningMethodHMAC); !ok {
			return nil, errors.New("签名方法错误")
		}
		return secret, nil
	})
	if err != nil || !token.Valid {
		return 0, "", errors.New("无效 token")
	}
	claims, ok := token.Claims.(jwt.MapClaims)
	if !ok {
		return 0, "", errors.New("无效 claims")
	}
	uid := uint(claims["uid"].(float64))
	username, _ := claims["username"].(string)
	return uid, username, nil
}
```

### 2. JWT 中间件

**backend/internal/middleware/auth.go**
```go
package middleware

import (
	"net/http"
	"strings"

	"github.com/gin-gonic/gin"
	"todo-app/internal/auth"
)

func JWTAuth() gin.HandlerFunc {
	return func(c *gin.Context) {
		header := c.GetHeader("Authorization")
		if !strings.HasPrefix(header, "Bearer ") {
			c.AbortWithStatusJSON(http.StatusUnauthorized, gin.H{"error": "缺少 token"})
			return
		}
		tokenStr := strings.TrimPrefix(header, "Bearer ")
		uid, username, err := auth.ParseToken(tokenStr)
		if err != nil {
			c.AbortWithStatusJSON(http.StatusUnauthorized, gin.H{"error": err.Error()})
			return
		}
		c.Set("uid", uid)
		c.Set("username", username)
		c.Next()
	}
}
```

### 3. 认证 Handler

**backend/internal/handler/auth.go**
```go
package handler

import (
	"net/http"

	"github.com/gin-gonic/gin"
	"todo-app/internal/auth"
	"todo-app/internal/store"
)

type AuthHandler struct {
	store *store.Store
}

func NewAuthHandler(s *store.Store) *AuthHandler {
	return &AuthHandler{store: s}
}

func (h *AuthHandler) Register(r *gin.Engine) {
	g := r.Group("/api/auth")
	g.POST("/register", h.RegisterUser)
	g.POST("/login", h.Login)
}

type credentials struct {
	Username string `json:"username" binding:"required,min=3"`
	Password string `json:"password" binding:"required,min=6"`
}

func (h *AuthHandler) RegisterUser(c *gin.Context) {
	var body credentials
	if err := c.ShouldBindJSON(&body); err != nil {
		c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
		return
	}

	hash, _ := auth.HashPassword(body.Password)
	u, err := h.store.CreateUser(body.Username, hash)
	if err != nil {
		c.JSON(http.StatusConflict, gin.H{"error": "用户名已存在"})
		return
	}

	token, _ := auth.GenerateToken(u.ID, u.Username)
	c.JSON(http.StatusCreated, gin.H{"token": token, "user": u})
}

func (h *AuthHandler) Login(c *gin.Context) {
	var body credentials
	if err := c.ShouldBindJSON(&body); err != nil {
		c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
		return
	}

	u, ok := h.store.FindUserByUsername(body.Username)
	if !ok || !auth.CheckPassword(u.Password, body.Password) {
		c.JSON(http.StatusUnauthorized, gin.H{"error": "用户名或密码错误"})
		return
	}

	token, _ := auth.GenerateToken(u.ID, u.Username)
	c.JSON(http.StatusOK, gin.H{"token": token, "user": u})
}
```

### 4. 改造 Todo Handler（加用户隔离）

**backend/internal/handler/todo.go**
```go
package handler

import (
	"net/http"
	"strconv"

	"github.com/gin-gonic/gin"
	"todo-app/internal/store"
)

type TodoHandler struct {
	store *store.Store
}

func NewTodoHandler(s *store.Store) *TodoHandler {
	return &TodoHandler{store: s}
}

func (h *TodoHandler) Register(r *gin.Engine) {
	g := r.Group("/api/todos")
	g.GET("", h.List)
	g.POST("", h.Create)
	g.PATCH("/:id/toggle", h.Toggle)
	g.DELETE("/:id", h.Delete)
}

func (h *TodoHandler) List(c *gin.Context) {
	uid := c.GetUint("uid")
	c.JSON(http.StatusOK, h.store.ListTodos(uid))
}

func (h *TodoHandler) Create(c *gin.Context) {
	uid := c.GetUint("uid")
	var body struct {
		Title string `json:"title" binding:"required"`
	}
	if err := c.ShouldBindJSON(&body); err != nil {
		c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
		return
	}
	c.JSON(http.StatusCreated, h.store.CreateTodo(uid, body.Title))
}

func (h *TodoHandler) Toggle(c *gin.Context) {
	uid := c.GetUint("uid")
	id, _ := strconv.Atoi(c.Param("id"))
	t, ok := h.store.ToggleTodo(uid, uint(id))
	if !ok {
		c.JSON(http.StatusNotFound, gin.H{"error": "not found"})
		return
	}
	c.JSON(http.StatusOK, t)
}

func (h *TodoHandler) Delete(c *gin.Context) {
	uid := c.GetUint("uid")
	id, _ := strconv.Atoi(c.Param("id"))
	if !h.store.DeleteTodo(uid, uint(id)) {
		c.JSON(http.StatusNotFound, gin.H{"error": "not found"})
		return
	}
	c.Status(http.StatusNoContent)
}
```

```ts
import axios from 'axios'

const http = axios.create({ baseURL: '/api' })

// 请求带上 token
http.interceptors.request.use(config => {
  const token = localStorage.getItem('token')
  if (token) {
    config.headers.Authorization = `Bearer ${token}`
  }
  return config
})

// 401 自动跳登录
http.interceptors.response.use(
  res => res,
  err => {
    if (err.response?.status === 401) {
      localStorage.removeItem('token')
      window.location.href = '/login'
    }
    return Promise.reject(err)
  }
)

export default http
```

### 2. API 加 token 拦截

**app/lib/api/api.dart**
```dart
import 'package:dio/dio.dart';
import 'package:shared_preferences/shared_preferences.dart';
import 'dart:io' show Platform;
import '../models/todo.dart';

class Api {
  static String get baseUrl {
    if (Platform.isAndroid) return 'http://10.0.2.2:8080/api';
    return 'http://localhost:8080/api';
  }

  final _dio = Dio(BaseOptions(baseUrl: baseUrl));

  Api() {
    // 请求自动带 token
    _dio.interceptors.add(InterceptorsWrapper(
      onRequest: (options, handler) async {
        final prefs = await SharedPreferences.getInstance();
        final token = prefs.getString('token');
        if (token != null) {
          options.headers['Authorization'] = 'Bearer $token';
        }
        handler.next(options);
      },
    ));
  }

  // ---------- 认证 ----------
  Future<void> login(String username, String password) async {
    final res = await _dio.post('/auth/login',
        data: {'username': username, 'password': password});
    final prefs = await SharedPreferences.getInstance();
    await prefs.setString('token', res.data['token']);
  }

  Future<void> register(String username, String password) async {
    final res = await _dio.post('/auth/register',
        data: {'username': username, 'password': password});
    final prefs = await SharedPreferences.getInstance();
    await prefs.setString('token', res.data['token']);
  }

  Future<void> logout() async {
    final prefs = await SharedPreferences.getInstance();
    await prefs.remove('token');
  }

  Future<bool> isLoggedIn() async {
    final prefs = await SharedPreferences.getInstance();
    return prefs.getString('token') != null;
  }

  // ---------- 待办 ----------
  Future<List<Todo>> list() async {
    final res = await _dio.get('/todos');
    return (res.data as List).map((e) => Todo.fromJson(e)).toList();
  }

  Future<Todo> create(String title) async {
    final res = await _dio.post('/todos', data: {'title': title});
    return Todo.fromJson(res.data);
  }

  Future<Todo> toggle(int id) async {
    final res = await _dio.patch('/todos/$id/toggle');
    return Todo.fromJson(res.data);
  }

  Future<void> remove(int id) => _dio.delete('/todos/$id');
}
```

```yaml
    networks: [appnet]
    restart: unless-stopped

  postgres:
    image: postgres:16-alpine
    environment:
      POSTGRES_USER: postgres
      POSTGRES_PASSWORD: ${DB_PASSWORD:-postgres}
      POSTGRES_DB: todo
    volumes:
      - pgdata:/var/lib/postgresql/data
    networks: [appnet]
    restart: unless-stopped
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U postgres"]
      interval: 5s
      timeout: 3s
      retries: 5

  web-builder:
    build: ./web
    volumes:
      - web_dist:/output
    networks: [appnet]

volumes:
  pgdata:
  web_dist:

networks:
  appnet:
```

**关键点**：
- `DB_HOST=postgres` 用容器名，走 Docker 网络
- `healthcheck` 确保 postgres 就绪，配合后端重试
- 数据持久化在 `pgdata` 卷
- 密码、JWT 密钥从 `.env` 读，不硬编码

**.env**（本地，别提交）
```
DB_PASSWORD=your-strong-password
JWT_SECRET=your-random-secret
```

## 六、CI/CD 流水线

**.github/workflows/ci-cd.yml**
```yaml
name: CI/CD

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

env:
  REGISTRY: ghcr.io
  IMAGE_PREFIX: ${{ github.repository }}

jobs:
  # ---------- 后端：测试 + 构建 ----------
  backend:
    runs-on: ubuntu-latest
    services:
      postgres:
        image: postgres:16-alpine
        env:
          POSTGRES_PASSWORD: postgres
          POSTGRES_DB: todo_test
        ports: ["5432:5432"]
        options: >-
          --health-cmd pg_isready
          --health-interval 5s
          --health-timeout 3s
          --health-retries 5
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-go@v5
        with:
          go-version: '1.22'

      - name: 下载依赖
        working-directory: ./backend
        run: go mod download

      - name: 静态检查
        working-directory: ./backend
        run: go vet ./...

      - name: 测试
        working-directory: ./backend
        env:
          DB_HOST: localhost
          DB_USER: postgres
          DB_PASSWORD: postgres
          DB_NAME: todo_test
        run: go test ./...

      - name: 构建
        working-directory: ./backend
        run: go build -o /dev/null ./cmd/server

  # ---------- 前端：构建 ----------
  web:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: '20'
      - name: 安装依赖
        working-directory: ./web
        run: npm ci
      - name: 构建
        working-directory: ./web
        run: npm run build

  # ---------- 构建并推送镜像 ----------
  docker:
    needs: [backend, web]
    if: github.ref == 'refs/heads/main' && github.event_name == 'push'
    runs-on: ubuntu-latest
    permissions:
      contents: read
      packages: write
    steps:
      - uses: actions/checkout@v4

      - name: 登录镜像仓库
        uses: docker/login-action@v3
        with:
          registry: ${{ env.REGISTRY }}
          username: ${{ github.actor }}
          password: ${{ secrets.GITHUB_TOKEN }}

      - name: 构建并推送后端
        uses: docker/build-push-action@v5
        with:
          context: ./backend
          push: true
          tags: |
            ${{ env.REGISTRY }}/${{ env.IMAGE_PREFIX }}/backend:latest
            ${{ env.REGISTRY }}/${{ env.IMAGE_PREFIX }}/backend:${{ github.sha }}

      - name: 构建并推送 Web
        uses: docker/build-push-action@v5
        with:
          context: ./web
          push: true
          tags: |
            ${{ env.REGISTRY }}/${{ env.IMAGE_PREFIX }}/web:latest
            ${{ env.REGISTRY }}/${{ env.IMAGE_PREFIX }}/web:${{ github.sha }}

  # ---------- 部署 ----------
  deploy:
    needs: docker
    runs-on: ubuntu-latest
    steps:
      - name: 部署到服务器
        uses: appleboy/ssh-action@v1
        with:
          host: ${{ secrets.SERVER_HOST }}
          username: ${{ secrets.SERVER_USER }}
          key: ${{ secrets.SERVER_SSH_KEY }}
          script: |
            cd /opt/todo-app
            docker compose pull
            docker compose up -d
            docker image prune -f
```

**流程**：push → 后端测试 → 前端构建 → 打镜像推仓库 → SSH 部署。

需要配置的 Secrets：
| Secret | 说明 |
|---|---|
| `SERVER_HOST` | 服务器 IP |
| `SERVER_USER` | SSH 用户 |
| `SERVER_SSH_KEY` | SSH 私钥 |
| `GITHUB_TOKEN` | 自动提供 |

## 九、整体架构（最终版）

```
Flutter (iOS/Android/Win) ┐
                          ├─→ Nginx → Go (JWT鉴权) → PostgreSQL
Vue (Web) ────────────────┘
                             ↑
                        Docker Compose 编排
                             ↑
                        GitHub Actions CI/CD
```

| 层 | 技术 | 能力 |
|---|---|---|
| 客户端 | Flutter + Vue | 多端 + Web |
| 入口 | Nginx | 反代、HTTPS、静态托管 |
| 后端 | Go + Gin | API、JWT 认证 |
| 数据 | PostgreSQL | 持久化、用户隔离 |
| 编排 | Docker Compose | 一键部署 |
| CI/CD | GitHub Actions | 测试、构建、推送、部署 |
