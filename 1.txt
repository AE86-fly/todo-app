### backend/internal/store/store.go
```go
package store

import (
	"sync"
	"todo-app/internal/model"
)

type Store struct {
	mu     sync.RWMutex
	todos  map[int]model.Todo
	nextID int
}

func New() *Store {
	return &Store{todos: make(map[int]model.Todo), nextID: 1}
}

func (s *Store) List() []model.Todo {
	s.mu.RLock()
	defer s.mu.RUnlock()
	list := make([]model.Todo, 0, len(s.todos))
	for _, t := range s.todos {
		list = append(list, t)
	}
	return list
}

func (s *Store) Create(title string) model.Todo {
	s.mu.Lock()
	defer s.mu.Unlock()
	t := model.Todo{ID: s.nextID, Title: title, Done: false}
	s.todos[s.nextID] = t
	s.nextID++
	return t
}

func (s *Store) Toggle(id int) (model.Todo, bool) {
	s.mu.Lock()
	defer s.mu.Unlock()
	t, ok := s.todos[id]
	if !ok {
		return model.Todo{}, false
	}
	t.Done = !t.Done
	s.todos[id] = t
	return t, true
}

func (s *Store) Delete(id int) bool {
	s.mu.Lock()
	defer s.mu.Unlock()
	if _, ok := s.todos[id]; !ok {
		return false
	}
	delete(s.todos, id)
	return true
}
```

### backend/internal/handler/todo.go
```go
package handler

import (
	"net/http"
	"strconv"

	"github.com/gin-gonic/gin"
	"todo-app/internal/model"
	"todo-app/internal/store"
)

type TodoHandler struct {
	store *store.Store
}

func NewTodoHandler(s *store.Store) *TodoHandler {
	return &TodoHandler{store: s}
}

func (h *TodoHandler) Register(r *gin.Engine) {
	g := r.Group("/api/todos")
	g.GET("", h.List)
	g.POST("", h.Create)
	g.PATCH("/:id/toggle", h.Toggle)
	g.DELETE("/:id", h.Delete)
}

func (h *TodoHandler) List(c *gin.Context) {
	c.JSON(http.StatusOK, h.store.List())
}

func (h *TodoHandler) Create(c *gin.Context) {
	var body struct {
		Title string `json:"title" binding:"required"`
	}
	if err := c.ShouldBindJSON(&body); err != nil {
		c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
		return
	}
	c.JSON(http.StatusCreated, h.store.Create(body.Title))
}

func (h *TodoHandler) Toggle(c *gin.Context) {
	id, _ := strconv.Atoi(c.Param("id"))
	t, ok := h.store.Toggle(id)
	if !ok {
		c.JSON(http.StatusNotFound, gin.H{"error": "not found"})
		return
	}
	c.JSON(http.StatusOK, t)
}

func (h *TodoHandler) Delete(c *gin.Context) {
	id, _ := strconv.Atoi(c.Param("id"))
	if !h.store.Delete(id) {
		c.JSON(http.StatusNotFound, gin.H{"error": "not found"})
		return
	}
	c.Status(http.StatusNoContent)
}
```

import (

```ts
import axios from 'axios'

const api = axios.create({ baseURL: '/api' })

export interface Todo {
  id: number
  title: string
  done: boolean
}

export const todoApi = {
  list: () => api.get<Todo[]>('/todos').then(r => r.data),
  create: (title: string) => api.post<Todo>('/todos', { title }).then(r => r.data),
  toggle: (id: number) => api.patch<Todo>(`/todos/${id}/toggle`).then(r => r.data),
  remove: (id: number) => api.delete(`/todos/${id}`),
}
```

### web/src/stores/todo.ts
```ts
import { defineStore } from 'pinia'
import { ref } from 'vue'
import { todoApi, type Todo } from '../api/todo'

export const useTodoStore = defineStore('todo', () => {
  const todos = ref<Todo[]>([])
  const loading = ref(false)

  async function fetch() {
    loading.value = true
    todos.value = await todoApi.list()
    loading.value = false
  }

  async function add(title: string) {
    if (!title.trim()) return
    todos.value.push(await todoApi.create(title))
  }

  async function toggle(id: number) {
    const t = await todoApi.toggle(id)
    const i = todos.value.findIndex(x => x.id === id)
    if (i >= 0) todos.value[i] = t
  }

  async function remove(id: number) {
    await todoApi.remove(id)
    todos.value = todos.value.filter(t => t.id !== id)
  }

  return { todos, loading, fetch, add, toggle, remove }
})
```

### app/lib/api/api.dart
```dart
import 'package:dio/dio.dart';
import 'dart:io' show Platform;
import '../models/todo.dart';

class Api {
  // Android 模拟器访问宿主机用 10.0.2.2，其余用 localhost
  static String get baseUrl {
    if (Platform.isAndroid) return 'http://10.0.2.2:8080/api';
    return 'http://localhost:8080/api';
  }

  final _dio = Dio(BaseOptions(baseUrl: baseUrl));

  Future<List<Todo>> list() async {
    final res = await _dio.get('/todos');
    return (res.data as List).map((e) => Todo.fromJson(e)).toList();
  }

  Future<Todo> create(String title) async {
    final res = await _dio.post('/todos', data: {'title': title});
    return Todo.fromJson(res.data);
  }

  Future<Todo> toggle(int id) async {
    final res = await _dio.patch('/todos/$id/toggle');
    return Todo.fromJson(res.data);
  }

  Future<void> remove(int id) => _dio.delete('/todos/$id');
}
```

### app/lib/pages/todo_page.dart
```dart
import 'package:flutter/material.dart';
import '../api/api.dart';
import '../models/todo.dart';

class TodoPage extends StatefulWidget {
  const TodoPage({super.key});

  @override
  State<TodoPage> createState() => _TodoPageState();
}

class _TodoPageState extends State<TodoPage> {
  final _api = Api();
  final _controller = TextEditingController();
  List<Todo> _todos = [];
  bool _loading = true;

  @override
  void initState() {
    super.initState();
    _load();
  }

  Future<void> _load() async {
    setState(() => _loading = true);
    _todos = await _api.list();
    setState(() => _loading = false);
  }

  Future<void> _add() async {
    final title = _controller.text.trim();
    if (title.isEmpty) return;
    final t = await _api.create(title);
    setState(() => _todos.add(t));
    _controller.clear();
  }

  Future<void> _toggle(int id) async {
    final t = await _api.toggle(id);
    setState(() {
      final i = _todos.indexWhere((x) => x.id == id);
      if (i >= 0) _todos[i] = t;
    });
  }

  Future<void> _remove(int id) async {
    await _api.remove(id);
    setState(() => _todos.removeWhere((t) => t.id == id));
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('待办事项')),
      body: Padding(
        padding: const EdgeInsets.all(16),
        child: Column(
          children: [
            Row(
              children: [
                Expanded(
                  child: TextField(
                    controller: _controller,
                    decoration: const InputDecoration(hintText: '输入待办...'),
                    onSubmitted: (_) => _add(),
                  ),
                ),
                IconButton(icon: const Icon(Icons.add), onPressed: _add),
              ],
            ),
            Expanded(
              child: _loading
                  ? const Center(child: CircularProgressIndicator())
                  : ListView.builder(
                      itemCount: _todos.length,
                      itemBuilder: (_, i) {
                        final t = _todos[i];
                        return ListTile(
                          leading: Checkbox(
                            value: t.done,
                            onChanged: (_) => _toggle(t.id),
                          ),
                          title: Text(
                            t.title,
                            style: TextStyle(
                              decoration: t.done
                                  ? TextDecoration.lineThrough
                                  : null,
                            ),
                          ),
                          trailing: IconButton(
                            icon: const Icon(Icons.delete),
                            onPressed: () => _remove(t.id),
                          ),
                        );
                      },
                    ),
            ),
          ],
        ),
      ),
    );
  }
}
```

## 五、Nginx 配置

### nginx/nginx.conf
```nginx
events { worker_connections 1024; }

http {
    include /etc/nginx/mime.types;

    server {
        listen 80;
        server_name _;

        # Vue 静态文件
        location / {
            root /usr/share/nginx/html;
            try_files $uri $uri/ /index.html;
        }

        # 后端 API
        location /api/ {
            proxy_pass http://backend:8080;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        }

        # 健康检查
        location /health {
            proxy_pass http://backend:8080/health;
        }
    }
}
```

## 六、Docker Compose 编排

### docker-compose.yml
```yaml
version: "3.8"

services:
  nginx:
    image: nginx:alpine
    ports:
      - "80:80"
    volumes:
      - ./nginx/nginx.conf:/etc/nginx/nginx.conf:ro
      - web_dist:/usr/share/nginx/html:ro
    depends_on:
      - backend
    networks: [appnet]
    restart: unless-stopped

  backend:
    build: ./backend
    environment:
      - PORT=8080
    networks: [appnet]
    restart: unless-stopped

  web-builder:
    build: ./web
    volumes:
      - web_dist:/output
    networks: [appnet]

volumes:
  web_dist:

networks:
  appnet:
```

**设计要点**：
- 只有 nginx 暴露 80，后端不占宿主端口 → **不会端口冲突**
- nginx 用容器名 `backend` 转发
- Vue 构建产物通过 `web_dist` 卷交给 nginx

## 七、启动步骤

### 1. 启动后端 + Web + Nginx
```bash
cd todo-app
docker compose up -d --build
```
访问 `http://localhost` 即可看到 Vue 页面，API 走 `/api`。

### 2. 单独开发后端（热重载）
```bash
cd backend
go run ./cmd/server
```

### 3. 单独开发 Vue（热更新）
```bash
cd web
npm install
npm run dev          # http://localhost:5173，代理到 8080
```

### 4. 运行 Flutter
```bash
cd app
flutter pub get
flutter run          # 选择设备：Windows / Android / iOS
```

**注意**：Flutter 在 Android 模拟器访问宿主机要用 `10.0.2.2`（代码里已处理）。

### 5. 打各端包
```bash
cd app
flutter build apk          # Android
flutter build ios          # iOS（需 Mac）
flutter build windows      # Windows 桌面
```
